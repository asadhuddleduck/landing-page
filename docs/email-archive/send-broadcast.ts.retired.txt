// Parse quoted CSV row (handles commas inside quotes)
function parseCsvRow(row: string): string[] {
  const fields: string[] = [];
  let current = "";
  let inQuotes = false;
  for (let i = 0; i < row.length; i++) {
    const ch = row[i];
    if (ch === '"') {
      inQuotes = !inQuotes;
    } else if (ch === "," && !inQuotes) {
      fields.push(current.trim());
      current = "";
    } else {
      current += ch;
    }
  }
  fields.push(current.trim());
  return fields;
}

const EMAIL_HEADERS = {
  "List-Unsubscribe": `<${UNSUB_MAILTO}>`,
};
const broadcastText = htmlToPlainText(broadcastHtml);

async function main() {
  const args = process.argv.slice(2);

  // --test: send a single test email
  if (args[0] === "--test") {
    const testEmail = args[1];
    if (!testEmail) {
      console.error(
        "Usage: npx tsx scripts/send-broadcast.ts --test you@email.com"
      );
      process.exit(1);
    }
    console.log(`Sending test email to ${testEmail}...`);
    const { error } = await resend.emails.send({
      from: marketingEmail.from,
      to: testEmail,
      replyTo: marketingEmail.replyTo,
      subject: "I built something for you",
      html: broadcastHtml,
      text: broadcastText,
      headers: EMAIL_HEADERS,
    });
    if (error) {
      console.error("Test send failed:", error);
      process.exit(1);
    }
    console.log("Test email sent! Check your inbox.");
    return;
  }

  // --send-all: loop through CSV and send individually, skipping agency clients
  if (args[0] === "--send-all") {
    const csvPath = resolve(
      process.env.HOME || "~",
      "Downloads/systeme export 28 feb 2026.csv"
    );
    const csv = readFileSync(csvPath, "utf-8");
    const lines = csv.split("\n").filter((l) => l.trim());
    const rows = lines.slice(1);

    let sent = 0;
    let skipped = 0;
    let failed = 0;

    for (let i = 0; i < rows.length; i++) {
      const fields = parseCsvRow(rows[i]);
      const email = fields[0];
      const tags = fields[7] || "";

      if (!email) {
        console.log(`[${i + 1}/${rows.length}] SKIP - no email`);
        skipped++;
        continue;
      }

      if (tags.includes("agency client")) {
        console.log(`[${i + 1}/${rows.length}] SKIP (client) - ${email}`);
        skipped++;
        continue;
      }

      const { error } = await resend.emails.send({
        from: marketingEmail.from,
        to: email,
        replyTo: marketingEmail.replyTo,
        subject: "I built something for you",
        html: broadcastHtml,
        text: broadcastText,
        headers: EMAIL_HEADERS,
      });

      if (error) {
        console.log(
          `[${i + 1}/${rows.length}] FAIL - ${email}: ${error.message}`
        );
        failed++;
      } else {
        console.log(`[${i + 1}/${rows.length}] SENT - ${email}`);
        sent++;
      }

      // Rate limit: 2 req/s max
      await new Promise((r) => setTimeout(r, 600));
    }

    console.log("---");
    console.log(`Done! ${sent} sent, ${skipped} skipped, ${failed} failed`);
    return;
  }

  // Default: show usage
  console.log("Usage:");
  console.log(
    "  npx tsx scripts/send-broadcast.ts --test you@email.com   # send test"
  );
  console.log(
    "  npx tsx scripts/send-broadcast.ts --send-all              # send to full list (excludes agency clients)"
  );
}

main().catch((err) => {
  console.error("Fatal error:", err);
  process.exit(1);
});
